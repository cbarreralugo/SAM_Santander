WITH base AS (
  SELECT t.*,
         REGEXP_REPLACE(NOMBRE, '¿+', 'Ñ') AS nombre_norm   -- 1) arregla '¿¿' -> 'Ñ'
  FROM tb_sam_usuarios_w t
)
SELECT ID_USUARIO,
       NOMBRE,
       USRLOCAL AS USRLOCAL_ACTUAL,
       LOWER(
         REGEXP_REPLACE(                                  -- 3) Ñ/ñ -> n
           REGEXP_SUBSTR(nombre_norm, '^\s*\S+') || '.' ||
           CASE
             WHEN REGEXP_COUNT(nombre_norm, '\S+') >= 3
               THEN REGEXP_SUBSTR(nombre_norm, '(\S+)\s+\S+\s*$', 1, 1, NULL, 1)
             ELSE REGEXP_SUBSTR(nombre_norm, '\S+$')
           END
         , '[Ññ]', 'n')
       ) AS USRLOCAL_PROPUESTO
FROM base
WHERE REGEXP_LIKE(USRLOCAL, '[0-9]');                     -- solo los que traen números
