SELECT CONCAT(
    'INSERT INTO [dbo].[IMSS_Trades] (',
    '[Fund],[InvNum],[TipoValor],[Td_Num],[CounterParty],[Buy_Sell],[TranType],',
    '[TradeFace],[OrigFace],[TradePrice],[TradeDate],[SettleDate],[Principal],[NetMoney],',
    '[Collateral_ISIN],[Collateral_Quantity],[Cupon],[Maturity],[IssueDate]) VALUES (',

    -- Fund
    CASE WHEN [Fund] IS NULL THEN 'NULL' ELSE '''' + REPLACE([Fund],'''','''''') + '''' END, ', ',
    -- InvNum
    CASE WHEN [InvNum] IS NULL THEN 'NULL' ELSE '''' + REPLACE([InvNum],'''','''''') + '''' END, ', ',
    -- TipoValor
    CASE WHEN [TipoValor] IS NULL THEN 'NULL' ELSE '''' + REPLACE([TipoValor],'''','''''') + '''' END, ', ',
    -- Td_Num (numeric)
    CASE WHEN [Td_Num] IS NULL THEN 'NULL' ELSE CONVERT(varchar(50), [Td_Num]) END, ', ',
    -- CounterParty
    CASE WHEN [CounterParty] IS NULL THEN 'NULL' ELSE '''' + REPLACE([CounterParty],'''','''''') + '''' END, ', ',
    -- Buy_Sell
    CASE WHEN [Buy_Sell] IS NULL THEN 'NULL' ELSE '''' + REPLACE([Buy_Sell],'''','''''') + '''' END, ', ',
    -- TranType
    CASE WHEN [TranType] IS NULL THEN 'NULL' ELSE '''' + REPLACE([TranType],'''','''''') + '''' END, ', ',

    -- Decimales
    CASE WHEN [TradeFace]  IS NULL THEN 'NULL' ELSE CONVERT(varchar(50), [TradeFace])  END, ', ',
    CASE WHEN [OrigFace]   IS NULL THEN 'NULL' ELSE CONVERT(varchar(50), [OrigFace])   END, ', ',
    CASE WHEN [TradePrice] IS NULL THEN 'NULL' ELSE CONVERT(varchar(50), [TradePrice]) END, ', ',

    -- Fechas como texto (tu esquema las define varchar)
    CASE WHEN [TradeDate]  IS NULL THEN 'NULL' ELSE '''' + REPLACE([TradeDate],'''','''''')  + '''' END, ', ',
    CASE WHEN [SettleDate] IS NULL THEN 'NULL' ELSE '''' + REPLACE([SettleDate],'''','''''') + '''' END, ', ',

    -- Más decimales
    CASE WHEN [Principal] IS NULL THEN 'NULL' ELSE CONVERT(varchar(50), [Principal]) END, ', ',
    CASE WHEN [NetMoney]  IS NULL THEN 'NULL' ELSE CONVERT(varchar(50), [NetMoney])  END, ', ',

    -- Texto / bigint
    CASE WHEN [Collateral_ISIN]     IS NULL THEN 'NULL' ELSE '''' + REPLACE([Collateral_ISIN],'''','''''') + '''' END, ', ',
    CASE WHEN [Collateral_Quantity] IS NULL THEN 'NULL' ELSE CONVERT(varchar(50), [Collateral_Quantity]) END, ', ',
    CASE WHEN [Cupon]    IS NULL THEN 'NULL' ELSE '''' + REPLACE([Cupon],'''','''''')    + '''' END, ', ',
    CASE WHEN [Maturity] IS NULL THEN 'NULL' ELSE '''' + REPLACE([Maturity],'''','''''') + '''' END, ', ',
    CASE WHEN [IssueDate] IS NULL THEN 'NULL' ELSE '''' + REPLACE([IssueDate],'''','''''') + '''' END,
    ');'
) AS InsertLine
FROM [dbo].[IMSS_Trades];
